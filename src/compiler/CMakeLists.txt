cmake_minimum_required(VERSION 3.20)
project(snabf_planner LANGUAGES CXX)

find_package(highs 1.16 CONFIG REQUIRED)
add_library(snabf_pass4 STATIC pass4/LayoutContract.cpp pass4/RingSwitchPlan.cpp pass4/DynamicRingPlanner.cpp)
target_compile_features(snabf_pass4 PUBLIC cxx_std_20)
target_include_directories(snabf_pass4 PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(snabf_pass3 STATIC pass3/GadgetNoiseContract.cpp pass3/ParameterMILP.cpp pass3/PrimeInstantiator.cpp pass3/ConcreteVerifier.cpp pass3/SitePricing.cpp)
target_compile_features(snabf_pass3 PUBLIC cxx_std_20)
target_include_directories(snabf_pass3 PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(snabf_pass3 PRIVATE highs::highs)
find_path(MPFI_INCLUDE_DIR mpfi.h REQUIRED)
find_library(MPFI_LIB mpfi REQUIRED)
find_library(MPFR_LIB mpfr REQUIRED)
find_library(GMP_LIB gmp REQUIRED)
find_library(GMPXX_LIB gmpxx REQUIRED)
add_library(snabf_pass2 STATIC pass2/PolyTemplates.cpp pass2/NonlinearOpRegistry.cpp pass2/IntervalCertifier.cpp pass2/CompositePolyDP.cpp pass2/NoiseTransfer.cpp)
target_compile_features(snabf_pass2 PUBLIC cxx_std_20)
target_include_directories(snabf_pass2 PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${MPFI_INCLUDE_DIR}")
target_link_libraries(snabf_pass2 PUBLIC "${MPFI_LIB}" "${MPFR_LIB}" "${GMP_LIB}")
# Both the SDK-free noise verifier and the full N1 library use these same objects.
add_library(snabf_scale_math OBJECT pass2/AffineErrorForm.cpp ir/ExactScaleTracker.cpp)
target_link_libraries(snabf_scale_math PUBLIC snabf_pass2 snabf_pass3 "${GMPXX_LIB}")
add_library(snabf_fullslot STATIC pass4/FullSlotProof.cpp)
target_link_libraries(snabf_fullslot PUBLIC snabf_scale_math)
add_library(snabf_noise STATIC ir/NoiseCertVerifier.cpp ir/BoundedNoiseTransfer.cpp)
add_library(snabf_affine_interval STATIC ir/AffineIntervalAnalyzer.cpp)
target_link_libraries(snabf_affine_interval PUBLIC snabf_pass2)

add_library(snabf_planner Orchestrator.cpp RouteSelection.cpp
  pass1/SubgroupTower.cpp pass1/KeyReuseModel.cpp pass1/RingSwitch.cpp)
target_compile_features(snabf_planner PUBLIC cxx_std_20)
target_include_directories(snabf_planner PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(snabf_planner PRIVATE highs::highs)

find_package(Boost REQUIRED)
add_executable(snabf-pass1 pass1/Planner.cpp)
target_include_directories(snabf-pass1 PRIVATE ${Boost_INCLUDE_DIRS})
target_link_libraries(snabf-pass1 PRIVATE snabf_planner)

include(CTest)
set(SNABF_MLIR_ROOT "" CACHE PATH "LLVM/MLIR SDK prefix (usr/lib/llvm-24)")
if(SNABF_MLIR_ROOT)
  add_library(snabf_exact_scale STATIC ir/ExactScaleEncoding.cpp)
  target_include_directories(snabf_exact_scale SYSTEM PRIVATE "${SNABF_MLIR_ROOT}/include")
  target_link_libraries(snabf_exact_scale PUBLIC snabf_scale_math PRIVATE "${SNABF_MLIR_ROOT}/lib/libLLVM.so")
  target_link_libraries(snabf_noise PUBLIC snabf_exact_scale)
  # LLVM's JSON reader and SHA256 implement the optional checked file format.
  # The planner without the SDK keeps its original dependency footprint.
  add_library(snabf_cost_profile STATIC pass3/CostProfile.cpp)
  target_include_directories(snabf_cost_profile SYSTEM PUBLIC "${SNABF_MLIR_ROOT}/include")
  target_link_libraries(snabf_cost_profile PUBLIC snabf_pass3 "${SNABF_MLIR_ROOT}/lib/libLLVM.so")
  add_subdirectory(ir)
else()
  target_link_libraries(snabf_noise PUBLIC snabf_scale_math)
endif()
