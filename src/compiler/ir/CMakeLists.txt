set(sdk "${SNABF_MLIR_ROOT}")
set(gen "${CMAKE_CURRENT_BINARY_DIR}")
include(HeirDialect.cmake)
foreach(item "Dialect;-dialect" "Types;-typedef" "Attrs;-attrdef" "Ops;-op")
  list(GET item 0 name)
  list(GET item 1 mode)
  foreach(ext h cpp)
    if(ext STREQUAL "h")
      set(action decls)
    else()
      set(action defs)
    endif()
    set(out "${gen}/${name}.${ext}.inc")
    add_custom_command(OUTPUT "${out}" COMMAND "${sdk}/bin/mlir-tblgen" "-gen${mode}-${action}"
      -I "${sdk}/include" -I "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/SNABF.td" -o "${out}"
      DEPENDS SNABF.td SNABFDialect.td SNABFTypes.td SNABFOps.td VERBATIM)
    list(APPEND generated "${out}")
  endforeach()
endforeach()
add_library(snabf_ir STATIC IR.cpp NativeContracts.cpp ${generated})
